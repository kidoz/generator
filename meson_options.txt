# Meson Build Options for Generator

# UI Backend (required - choose one)
option('ui-backend',
  type: 'combo',
  choices: ['gtk4', 'console'],
  value: 'gtk4',
  description: 'User interface backend to build (gtk4=GTK4/libadwaita, console=SDL3 only)'
)

# Z80 Emulator (required - choose one)
option('z80-backend',
  type: 'combo',
  choices: ['cmz80', 'raze'],
  value: 'cmz80',
  description: 'Z80 emulator implementation (cmz80=portable C, raze=x86 assembly, requires nasm)'
)

# Optional features
option('logging',
  type: 'boolean',
  value: true,
  description: 'Enable logging output'
)

option('gcc-version',
  type: 'combo',
  choices: ['2', '3', 'auto', 'no'],
  value: 'auto',
  description: 'GCC optimization version (auto=detect, no=disable optimizations)'
)

option('netplay',
  type: 'boolean',
  value: true,
  description: 'Enable Kaillera network multiplayer support'
)

# Audio quality options
option('audio-rate',
  type: 'combo',
  choices: ['44100', '48000', '96000'],
  value: '48000',
  description: 'Audio sample rate in Hz (48000 recommended for modern systems)'
)

option('audio-oversampling',
  type: 'combo',
  choices: ['1', '2', '4'],
  value: '2',
  description: 'Internal oversampling factor for anti-aliasing (2x recommended)'
)

option('audio-hq-filter',
  type: 'boolean',
  value: true,
  description: 'Enable high-quality multi-stage audio filtering'
)
